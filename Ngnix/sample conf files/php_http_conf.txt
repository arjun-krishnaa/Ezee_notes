server {
    listen 80;
    server_name kannabiran.com;
    return 404; # managed by Certbot

    root /var/www/html/ezeeinfo/ezeeinfo;
    index index.html index.htm;

    access_log /var/log/nginx/host.access.log main;

    location / {
        try_files $uri $uri/ =404;
    }

    location /.well-known/acme-challenge/ {
        root /var/www/html;
        allow all;
    }

    error_page 404 /404.html;

    error_page 500 502 503 504 /50x.html;
    location = /50x.html {
        root /usr/share/nginx/html;
    }
}

server {
    location ~ \.php$ {
        root html;
        fastcgi_pass 127.0.0.1:9000;
        fastcgi_index index.php;
        fastcgi_param SCRIPT_FILENAME /scripts$fastcgi_script_name;
        include fastcgi_params;
    }

    location /logs {
        alias /var/log/nginx;
        autoindex on;
        autoindex_exact_size off;
        autoindex_localtime on;
        default_type text/plain;

        auth_basic "Restricted Log Access";
        auth_basic_user_file /etc/nginx/.jenkins_logs_auth;

        limit_except GET HEAD {
        }
    }

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/kannabiran.com/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/kannabiran.com/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}
