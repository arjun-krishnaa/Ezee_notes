    listen      80 default_server;  --------------------------------------------------------------------------> Listens on port 80 (HTTP) & [default_server → this is the fallback server if no match is found]

    location /busservices {   --------------------------------------------------------------------------------> Matches requests like - [http://app.r3.ezeebits.com/busservices]
        proxy_redirect          off;  ----------------------------------------------------------------------------> Prevents NGINX from modifying redirects from backend
        proxy_set_header        Host               $host;   -------------------------------------------------> Passes original host header to backend
        proxy_set_header        X-Real-IP          $remote_addr;  ------------------------------------> Sends client real IP
        proxy_set_header        X-Forwarded-For    $proxy_add_x_forwarded_for; ---------> Adds client IP chain (important for logging)
        proxy_set_header        X-Forwarded-Proto  http; --------------------------------------------> Tells backend request came via HTTP
        proxy_pass              http://localhost:8080; ----------------------------------------------------> Forwards request to: Tomcat / App running on port 8080 
        proxy_buffer_size       8k;   --------------------------------------------------------------------------> Controls response buffering, Helps performance & large responses
        proxy_buffers           16 32k; ------------------------------------------------------------------------> Controls response buffering, Helps performance & large responses
        proxy_intercept_errors on;  -------------------------------------------------------------------------> NGINX handles errors instead of backend
        proxy_connect_timeout       600; -------------------------------------------------------------------> Very high timeout (600 sec = 10 min) [Useful for: Long API calls, File uploads]
        proxy_send_timeout          600;  -------------------------------------------------------------------> Very high timeout (600 sec = 10 min) [Useful for: Long API calls, File uploads]
        proxy_read_timeout          600;   -------------------------------------------------------------------> Very high timeout (600 sec = 10 min) [Useful for: Long API calls, File uploads]

    }
