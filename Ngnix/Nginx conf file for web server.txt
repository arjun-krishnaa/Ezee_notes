# Proxy to Java service
 location ^~ /cargoservices/ {
    proxy_pass http://cargo_java/cargoservices/;
    proxy_http_version 1.1;

    proxy_set_header Connection "";
    proxy_set_header Host dev-app.ezeecargo.in;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;

    proxy_connect_timeout 2s;
    proxy_send_timeout 15s;
    proxy_read_timeout 15s;

    proxy_buffering on;
    proxy_buffers 16 16k;
    proxy_busy_buffers_size 32k;
 }

    access_log /var/log/nginx/portal.access.log custom_json;

    ssl_certificate /usr/letsencrypt/webroot/config/live/dev-ui-v3.ezeecargo.in/fullchain.pem;
    ssl_certificate_key /usr/letsencrypt/webroot/config/live/dev-ui-v3.ezeecargo.in/privkey.pem;



    # Logging directory
    location /logs/ {
        alias /var/www/cargo/cargo-console/webapp/writable/logs/;
        autoindex on;
        autoindex_exact_size off;
        autoindex_localtime on;
        default_type text/plain;

        auth_basic "Restricted Log Access";
        auth_basic_user_file /etc/nginx/.htpasswd;

        limit_except GET HEAD { deny all; }
    }

    include /etc/nginx/global/php.conf;

    # Include other global configs
    include /etc/nginx/global/letsencrypt-acme-challenge.conf;
    include /etc/nginx/global/restrictions.conf;
}
